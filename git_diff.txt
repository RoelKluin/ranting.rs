+struct Ask {
+    oracle: Expr,
+    question: String,
+}
+
+#[proc_macro]
+pub fn ask(input: TokenStream1) -> TokenStream1 {
+    let ask = syn::parse_macro_input!(input as Ask);
+    let oracle = ask.oracle;
+    let question = ask.question;
+    let answer: TokenStream = parse_quote!(#oracle.answer(#question));
+    answer.into()
+}
+
+impl syn::parse::Parse for Ask {
+    fn parse(input: syn::parse::ParseStream) -> syn::Result<Self> {
+        lazy_static! {
+            static ref ASK: Regex = Regex::new(lang::ASK).unwrap();
+        }
+        if input.is_empty() {
+            return Err(Error::new(Span::mixed_site(), "missing question"));
+        }
+        let oracle = input.parse::<Expr>()?;
+        input.parse::<Token![,]>()?;
+        let lit = input.parse::<StrLit>()?;
+        let src = lit.to_slice();
+        let text = src.text();
+        #[cfg(feature = "debug")]
+        eprintln!("{}", question);
+        let mut err: Option<(usize, usize, String)> = None;
+        let question = ASK
+            .replace_all(text, |caps: &Captures| String::new())
+            .to_string();
+        // XXX there was no err assignment, was this intended?
+        if question.is_empty() {
+            err = Some((0, text.len(), "Error in question".to_string()));
+        }
+        match err {
+            Some((start, end, msg)) => Err(src.slice(start..end).error(msg.as_str())),
+            None => Ok(Ask { oracle, question }),
+        }
+    }
+}
